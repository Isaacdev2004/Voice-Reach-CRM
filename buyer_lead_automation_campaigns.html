// ═══════════════════════════════════════════════════════
// CAMPAIGN 2: IDX Listing Alert Drip
// Platform: KVCore / Follow Up Boss / Zapier + Mailchimp
// Trigger:  New MLS listing matches saved buyer search
// ═══════════════════════════════════════════════════════

const listingAlertDrip = {

  // ── TRIGGER: New listing matches saved search ─────────
  trigger: {
    type: "mls_new_listing_event",
    source: "IDX feed (RETS / RESO Web API)",
    condition: "listing.match_score >= 70",  // 0–100 relevance score
    fields_matched: ["bedrooms", "bathrooms", "price_range", "zip_codes", "property_type"]
  },

  // ── INSTANT SMS (match score > 90 only) ──────────────
  instantSMS: {
    condition: "listing.match_score > 90 AND lead.sms_consent == true",
    delay: 0,
    action: "send_sms",
    throttle: "max_1_sms_per_day_per_lead",
    body: "{{first_name}}, new match! {{listing_address}} — {{beds}}bd/{{baths}}ba at {{price}}..."
  },

  // ── WEEKLY EMAIL DIGEST ───────────────────────────────
  weeklyDigest: {
    schedule: "CRON: 0 8 * * 1",   // Every Monday at 8:00 AM
    condition: "lead.stage NOT IN ['Under Contract','Closed','Paused','Unsubscribed']",
    action: "send_email",
    template: "weekly_listing_digest",
    data_sources: {
      listings:   "mls_api.get_new_listings(lead.saved_search, days=7)",
      avg_dom:    "mls_api.avg_days_on_market(lead.zip_codes)",
      median_price: "mls_api.median_list_price(lead.zip_codes)",
    },
    skip_if: "new_listings_count == 0"
  },

  // ── PRICE DROP ALERT ──────────────────────────────────
  priceDropAlert: {
    trigger: "mls_price_reduction_event on saved_listings",
    delay: 0,
    action: "send_sms",
    condition: "price_drop_pct >= 2 AND lead.sms_consent == true",
    body: "{{first_name}}, price drop alert on a home you saved!..."
  },

  // ── MONTHLY CHECK-IN ──────────────────────────────────
  monthlyCheckIn: {
    schedule: "day 30, 60, 90, 120 after registration",
    condition: "lead.showing_not_scheduled AND lead.active",
    action: "send_email",
    template: "monthly_checkin_with_market_update"
  },

  // ── RE-ENGAGEMENT (no opens in 60 days) ───────────────
  reEngagement: {
    condition: "lead.last_email_open > 60 days ago",
    action: "send_sms",
    body: "Hey {{first_name}}, still looking in {{city}}? Happy to adjust your search or pause alerts. Just reply YES to continue or PAUSE to take a break."
  },

  // ── EXIT CONDITIONS ───────────────────────────────────
  exitConditions: [
    "lead.showing_booked == true",
    "lead.replied 'not looking'",
    "lead.stage IN ['Under Contract','Closed']",
    "lead.unsubscribed == true"
  ],

  // ── RECOMMENDED IDX PLATFORMS ────────────────────────
  platforms: ["KVCore (built-in)", "Showcase IDX + Zapier", "iHomefinder", "Homes.com"]
};
IDX IntegrationConnect your MLS IDX feed to your CRM's saved search feature. KVCore and Follow Up Boss handle this natively. For standalone IDX (Showcase IDX, iHomefinder), use Zapier to sync new match events.
Match scoringSet instant SMS threshold to 90%+ match score to avoid over-texting. Email digests can include 70%+ matches. Most IDX platforms expose a match score field you can reference in your automation condition.
Frequency capNever send more than 1 listing SMS per day per lead, regardless of how many new matches come in. Batch them into the Monday digest instead.

Post-showing follow-up logic
// ═══════════════════════════════════════════════════════
// CAMPAIGN 5: Post-Showing Follow-Up & Offer Push
// Trigger:  Agent logs showing as "completed" in CRM
//           OR calendar event with tag "showing" ends
// Platform: Follow Up Boss / KVCore / HubSpot + Zapier
// ═══════════════════════════════════════════════════════

const postShowingFlow = {

  trigger: {
    type: "crm_event",
    event: "showing_status updated to 'completed'",
    or: "calendar event tagged 'showing' ends",
    required_fields: ["property_address", "lead_id", "showing_date"]
  },

  // ── STEP 1: Same-day feedback SMS ────────────────────
  step1: {
    delay: 7200,     // 2 hours after showing_end_time
    action: "send_sms",
    body: "Hey {{first_name}}! Hope the showing was helpful. Quick question — overall: loved it, liked it, or not the one?",
    // Parse reply to update CRM stage:
    reply_routing: {
      "loved":    "stage = 'Hot Buyer' → expedite sequence",
      "liked":    "stage = 'Warm Buyer' → standard sequence",
      "not":      "stage = 'Active Searching' → return to listing alert drip"
    }
  },

  // ── STEP 2: Comp analysis email (Day 1) ──────────────
  step2: {
    delay: 86400,    // 24 hours
    action: "send_email",
    template: "post_showing_comp_analysis",
    subject: "The numbers on {{property_address}} — is it priced right?",
    data_fetch: {
      comps: "mls_api.get_sold_comps(property_address, radius=0.5mi, days=90)",
      avg_ppsf: "mls_api.avg_price_per_sqft(neighborhood)",
      suggested_offer: "calc: avg_ppsf * listing_sqft * agent_adjustment_factor"
    }
  },

  // ── STEP 3: Offer nudge SMS (Day 2) ──────────────────
  step3: {
    delay: 172800,   // 48 hours
    action: "send_sms",
    condition: "lead.offer_submitted == false",
    body: "{{first_name}}, I ran the comps and sent you an analysis — did you get a chance to look?..."
  },

  // ── STEP 4: Competing offer alert (Day 3) ────────────
  step4: {
    delay: 259200,   // 72 hours
    action: "send_email",
    condition: "lead.offer_submitted == false",
    template: "competing_offer_urgency",
    data_fetch: {
      competing_offers: "agent_manually_inputs OR listing_agent_confirmed",
      showing_count:    "mls_api.get_showing_count(property_id)"
    }
  },

  // ── ON OFFER SUBMITTED ────────────────────────────────
  onOfferSubmitted: {
    trigger: "lead.offer_submitted == true",
    actions: [
      { action: "cancel_all_pending_campaign_steps" },
      { action: "update_stage", stage: "Offer Submitted" },
      { action: "send_sms", body: "Offer is in, {{first_name}}! I'll update you as soon as I hear back. Fingers crossed! 🤞 — {{agent_name}}" }
    ]
  },

  // ── OFFER REJECTED → RE-ENTER SEARCH ────────────────
  onOfferRejected: {
    actions: [
      { action: "send_sms", body: "{{first_name}}, offer wasn't accepted — I know that's disappointing. The right home is still out there. Let's regroup. When can we chat? — {{agent_name}}" },
      { action: "re-enter listing_alert_drip sequence" }
    ]
  },

  exitConditions: [
    "lead.offer_submitted == true",
    "lead.under_contract == true",
    "lead.feedback == 'not the one' → re-enter listing_alert_drip",
    "lead.unsubscribed == true"
  ]
};
